cmake_minimum_required(VERSION 3.20)
project(SnowLumaNativeRecovery LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This recovery project targets Windows x64 only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
add_compile_options(/utf-8 /W4)
add_library(injector SHARED src/addon.cpp src/windows_services.cpp src/pe_image.cpp src/loader.cpp src/remote_stub.cpp)
target_include_directories(injector PRIVATE include vendor/node-addon-api toolchain/node-sdk/include/node)
target_compile_definitions(injector PRIVATE NAPI_VERSION=8 NAPI_CPP_EXCEPTIONS)
target_link_libraries(injector PRIVATE "${CMAKE_SOURCE_DIR}/toolchain/node-sdk/Release/node.lib" Advapi32 Rstrtmgr)
set_target_properties(injector PROPERTIES PREFIX "" SUFFIX ".node" OUTPUT_NAME "snowluma-win32-x64")
set_source_files_properties(src/remote_stub.cpp PROPERTIES COMPILE_OPTIONS "/Od;/GS-;/Gy-;/guard:cf-;/Zl")
target_link_options(injector PRIVATE /OPT:NOICF /INCREMENTAL:NO)
add_library(resident SHARED src/resident.cpp)
target_include_directories(resident PUBLIC include)
set_target_properties(resident PROPERTIES OUTPUT_NAME snowluma_windows_resident_dispatch)
add_library(hook_core STATIC src/pipe_protocol.cpp src/pe_image.cpp src/message_snapshot.cpp)
target_include_directories(hook_core PUBLIC include)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE hook_core)
add_executable(resident_tests tests/resident_tests.cpp)
target_include_directories(resident_tests PRIVATE include)
add_executable(fixture_host tests/fixture_host.cpp)
add_library(fixture_module SHARED tests/fixture_module.cpp)
set_target_properties(fixture_module PROPERTIES OUTPUT_NAME fixture_module)
add_library(fixture_busy SHARED tests/fixture_module.cpp)
target_compile_definitions(fixture_busy PRIVATE FIXTURE_BUSY=1)
set_target_properties(fixture_busy PROPERTIES OUTPUT_NAME fixture_busy)
enable_testing()
add_test(NAME core COMMAND core_tests "$<TARGET_FILE:fixture_module>" "${CMAKE_SOURCE_DIR}/reference/hook.dll")
add_test(NAME resident_rebuilt COMMAND resident_tests "$<TARGET_FILE:resident>")
add_test(NAME resident_original COMMAND resident_tests "${CMAKE_SOURCE_DIR}/reference/resident.bin")
add_test(NAME resident_exceptions COMMAND resident_tests "$<TARGET_FILE:resident>" --exceptions)
